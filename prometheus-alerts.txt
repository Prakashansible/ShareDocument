---
# postgresql-alerting-rules.yaml
apiVersion: monitoring.coreos.com/v1
kind: PrometheusRule
metadata:
  name: postgresql-alerts
  namespace: postgresql
  labels:
    prometheus: kube-prometheus
    role: alert-rules
spec:
  groups:
  - name: postgresql.rules
    interval: 30s
    rules:
    # Database Availability
    - alert: PostgreSQLDown
      expr: up{job="postgres-exporter"} == 0
      for: 1m
      labels:
        severity: critical
        service: postgresql
      annotations:
        summary: "PostgreSQL instance is down"
        description: "PostgreSQL instance {{ $labels.instance }} has been down for more than 1 minute."

    # Connection Issues
    - alert: PostgreSQLTooManyConnections
      expr: pg_database_connections{job="postgres-exporter"} > 80
      for: 5m
      labels:
        severity: warning
        service: postgresql
      annotations:
        summary: "Too many connections to PostgreSQL"
        description: "PostgreSQL instance {{ $labels.instance }} has {{ $value }} connections (threshold: 80)."

    - alert: PostgreSQLMaxConnectionsReached
      expr: pg_database_connections{job="postgres-exporter"} >= pg_settings_max_connections{job="postgres-exporter"} * 0.9
      for: 5m
      labels:
        severity: critical
        service: postgresql
      annotations:
        summary: "PostgreSQL max connections reached"
        description: "PostgreSQL instance {{ $labels.instance }} has reached 90% of max connections."

    # Performance Issues
    - alert: PostgreSQLHighRollbackRate
      expr: rate(pg_database_rollbacks{job="postgres-exporter"}[5m]) / rate(pg_database_commits{job="postgres-exporter"}[5m]) > 0.1
      for: 5m
      labels:
        severity: warning
        service: postgresql
      annotations:
        summary: "High rollback rate detected"
        description: "PostgreSQL database {{ $labels.datname }} has a rollback ratio of {{ $value | humanizePercentage }}."

    - alert: PostgreSQLLowCacheHitRatio
      expr: pg_database_blocks_hit{job="postgres-exporter"} / (pg_database_blocks_hit{job="postgres-exporter"} + pg_database_blocks_read{job="postgres-exporter"}) < 0.9
      for: 10m
      labels:
        severity: warning
        service: postgresql
      annotations:
        summary: "Low cache hit ratio"
        description: "PostgreSQL database {{ $labels.datname }} has a cache hit ratio of {{ $value | humanizePercentage }} (< 90%)."

    - alert: PostgreSQLSlowQueries
      expr: pg_slow_queries{job="postgres-exporter"} > 5
      for: 5m
      labels:
        severity: warning
        service: postgresql
      annotations:
        summary: "Too many slow queries"
        description: "PostgreSQL instance {{ $labels.instance }} has {{ $value }} queries running for more than 1 minute."

    # Replication Issues
    - alert: PostgreSQLReplicationLag
      expr: pg_replication_lag{job="postgres-exporter"} > 10
      for: 5m
      labels:
        severity: warning
        service: postgresql
      annotations:
        summary: "PostgreSQL replication lag"
        description: "PostgreSQL replication lag is {{ $value }}s for instance {{ $labels.instance }}."

    # Storage Issues
    - alert: PostgreSQLDatabaseSizeTooLarge
      expr: pg_database_size_bytes{job="postgres-exporter"} > 100 * 1024 * 1024 * 1024  # 100GB
      for: 10m
      labels:
        severity: warning
        service: postgresql
      annotations:
        summary: "PostgreSQL database too large"
        description: "PostgreSQL database {{ $labels.datname }} size is {{ $value | humanize1024 }}B."

    # Maintenance Issues
    - alert: PostgreSQLTooManyDeadTuples
      expr: pg_table_stats_dead_tuples{job="postgres-exporter"} > 100000
      for: 10m
      labels:
        severity: warning
        service: postgresql
      annotations:
        summary: "Too
